"use client";

import { useEffect, useMemo, useState } from "react";

import { PageLayout } from "@/components/layout/PageLayout";
import { fetchFromAPI } from "@/lib/api";
import { getCurrentMonth } from "@/lib/period";
import { formatCurrency } from "@/lib/utils";

interface SubscriptionItem {
  original_name: string;
  date: string;
  amount: number;
}

interface SubscriptionsSummary {
  month: string;
  active_count: number;
  monthly_total: number;
  annual_projection: number;
  average_per_service: number;
  items: SubscriptionItem[];
}

function formatDate(dateValue: string) {
  const date = new Date(`${dateValue}T00:00:00`);
  if (Number.isNaN(date.getTime())) {
    return dateValue;
  }

  return date.toLocaleDateString("pt-BR", { day: "2-digit", month: "short", year: "numeric" });
}

function LoadingLine({ width = "w-full" }: { width?: string }) {
  return <div className={`h-5 ${width} animate-pulse rounded bg-white/[0.05]`} />;
}

export default function SubscriptionsPage() {
  const [summary, setSummary] = useState<SubscriptionsSummary | null>(null);
  const [loading, setLoading] = useState(true);

  const month = useMemo(() => getCurrentMonth(), []);

  useEffect(() => {
    let active = true;

    async function load() {
      setLoading(true);
      const payload = await fetchFromAPI<SubscriptionsSummary>(`/dashboard/subscriptions?month=${month}`);

      if (!active) {
        return;
      }

      setSummary(payload);
      setLoading(false);
    }

    void load();

    return () => {
      active = false;
    };
  }, [month]);

  const items = summary?.items ?? [];

  return (
    <PageLayout title="Assinaturas" subtitle={`Serviços recorrentes mapeados em ${month}.`} hideTitle={true}>
      <section className="grid gap-4 md:grid-cols-2 xl:grid-cols-4">
        {loading || !summary ? (
          Array.from({ length: 4 }).map((_, index) => (
            <article key={`sub-card-loading-${index}`} className="rounded-xl border border-white/[0.07] bg-[#1a1a1a] p-4">
              <LoadingLine width="w-28" />
              <div className="mt-3">
                <LoadingLine width="w-36" />
              </div>
            </article>
          ))
        ) : (
          [
            { label: "Assinaturas Ativas", value: String(summary.active_count) },
            { label: "Gasto Mensal", value: formatCurrency(summary.monthly_total) },
            { label: "Projeção Anual", value: formatCurrency(summary.annual_projection) },
            { label: "Média por Serviço", value: formatCurrency(summary.average_per_service) },
          ].map((item) => (
            <article key={item.label} className="rounded-xl border border-white/[0.07] bg-[#1a1a1a] p-4">
              <p className="text-xs uppercase tracking-widest text-neutral-500">{item.label}</p>
              <p className="mt-3 text-2xl font-semibold text-white">{item.value}</p>
            </article>
          ))
        )}
      </section>

      <section className="rounded-xl border border-white/[0.07] bg-[#1a1a1a] p-4">
        <div className="grid grid-cols-[2fr_1fr_1fr] gap-3 border-b border-white/[0.05] pb-3 text-xs uppercase tracking-widest text-neutral-600">
          <span>Serviço</span>
          <span className="text-right">Data</span>
          <span className="text-right">Valor Mensal</span>
        </div>

        {loading ? (
          <div className="mt-3 space-y-2">
            {Array.from({ length: 4 }).map((_, index) => (
              <LoadingLine key={`sub-list-loading-${index}`} width="w-full" />
            ))}
          </div>
        ) : items.length === 0 ? (
          <p className="py-8 text-sm text-neutral-500">Nenhuma assinatura identificada no mês selecionado.</p>
        ) : (
          <div className="mt-2">
            {items.map((item, index) => (
              <article
                key={`${item.original_name}-${item.date}-${index}`}
                className="grid grid-cols-[2fr_1fr_1fr] items-center gap-3 border-b border-white/[0.05] py-3 last:border-b-0"
              >
                <p className="truncate text-sm text-white">{item.original_name}</p>
                <p className="text-right text-sm text-neutral-400">{formatDate(item.date)}</p>
                <p className="text-right text-sm font-medium text-emerald-300">{formatCurrency(item.amount)}</p>
              </article>
            ))}
          </div>
        )}
      </section>
    </PageLayout>
  );
}

"use client";

import { Suspense, useEffect, useMemo, useState } from "react";
import {
  ArrowDownRight,
  ArrowRightLeft,
  ArrowUpRight,
  Building2,
  CalendarRange,
  ChevronLeft,
  ChevronRight,
  Filter,
  Save,
  Search,
  SlidersHorizontal,
  X,
} from "lucide-react";
import { usePathname, useRouter, useSearchParams } from "next/navigation";

import { Button } from "@/components/ui/button";
import { PageLayout } from "@/components/layout/PageLayout";
import { fetchFromAPI } from "@/lib/api";
import { formatCurrency } from "@/lib/utils";
import { getCurrentMonth, getMonthBounds } from "@/lib/period";

interface Transaction {
  id: string;
  date: string;
  description: string;
  category: string;
  natureza: string;
  subnatureza: string;
  account: string;
  amount: number | string;
}

interface TransactionListResponse {
  items: Transaction[];
  total: number;
  limit: number;
  offset: number;
  totalPages: number;
  summary: {
    income: number;
    expense: number;
    balance: number;
  };
}

interface ReclassificationPayload {
  macro_categoria: string;
  natureza: string;
  subnatureza: string;
}

interface CatalogResponse {
  categories: string[];
  natures: string[];
  accounts: string[];
}

const NATURE_LABELS: Record<string, string> = {
  INCOME: "Receita",
  EXPENSE: "Despesa",
  TRANSFER: "Transferência",
  INVESTMENT: "Investimento",
  DIVIDEND: "Dividendo",
};

const emptyPayload: ReclassificationPayload = {
  macro_categoria: "",
  natureza: "",
  subnatureza: "",
};

type TransactionKind = "income" | "expense" | "transfer" | "other";

function normalizeText(value: string) {
  return value
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .toUpperCase()
    .trim();
}

function classifyTransaction(natureza: string): TransactionKind {
  const normalized = normalizeText(natureza);

  if (["INCOME", "RECEITA", "CREDIT", "CREDITO", "CRÉDITO"].includes(normalized)) {
    return "income";
  }

  if (["EXPENSE", "DESPESA", "DEBIT", "DEBITO", "DÉBITO"].includes(normalized)) {
    return "expense";
  }

  if (["TRANSFER", "TRANSFERENCIA", "TRANSFERÊNCIA", "TRANSFERENCIA ENTRE CONTAS"].includes(normalized)) {
    return "transfer";
  }

  return "other";
}

function TransactionsPageBody() {
  const pathname = usePathname();
  const router = useRouter();
  const searchParams = useSearchParams();

  const fallbackMonth = getCurrentMonth();
  const fallbackMonthBounds = getMonthBounds(fallbackMonth);

  const [startDate, setStartDate] = useState(() => searchParams.get("startDate") ?? fallbackMonthBounds.startDate);
  const [endDate, setEndDate] = useState(() => searchParams.get("endDate") ?? fallbackMonthBounds.endDate);
  const [account, setAccount] = useState(() => searchParams.get("account") ?? "all");
  const [category, setCategory] = useState(() => searchParams.get("category") ?? "all");
  const [natureza, setNatureza] = useState(() => searchParams.get("natureza") ?? "all");
  const [limit] = useState(50);
  const [offset, setOffset] = useState(0);

  const [catalog, setCatalog] = useState<CatalogResponse>({
    accounts: [],
    categories: [],
    natures: [],
  });
  const [transactions, setTransactions] = useState<Transaction[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [searchTerm, setSearchTerm] = useState("");
  const [total, setTotal] = useState(0);
  const [totalPages, setTotalPages] = useState(0);
  const [summary, setSummary] = useState({
    income: 0,
    expense: 0,
    balance: 0,
  });
  const [selectedTransaction, setSelectedTransaction] = useState<Transaction | null>(null);
  const [form, setForm] = useState<ReclassificationPayload>(emptyPayload);
  const [saving, setSaving] = useState(false);
  const [message, setMessage] = useState<string | null>(null);

  const currentPage = Math.floor(offset / limit) + 1;
  const hasPreviousPage = offset > 0;
  const hasNextPage = totalPages > 0 && currentPage < totalPages;
  const rangeStart = total > 0 ? offset + 1 : 0;
  const rangeEnd = total > 0 ? Math.min(offset + transactions.length, total) : 0;

  useEffect(() => {
    const nextParams = new URLSearchParams();

    nextParams.set("startDate", startDate);
    nextParams.set("endDate", endDate);

    if (account !== "all") {
      nextParams.set("account", account);
    }

    if (category !== "all") {
      nextParams.set("category", category);
    }

    if (natureza !== "all") {
      nextParams.set("natureza", natureza);
    }

    const nextQuery = nextParams.toString();

    if (nextQuery !== searchParams.toString()) {
      router.replace(nextQuery ? `${pathname}?${nextQuery}` : pathname, { scroll: false });
    }
  }, [account, category, endDate, natureza, pathname, router, searchParams, startDate]);

  useEffect(() => {
    let active = true;

    async function loadCatalog() {
      const data = await fetchFromAPI<CatalogResponse>("/catalog");

      if (!active) {
        return;
      }

      if (data) {
        setCatalog(data);
      }
    }

    loadCatalog();

    return () => {
      active = false;
    };
  }, []);

  useEffect(() => {
    let active = true;

    async function loadData() {
      setLoading(true);
      setError(null);

      try {
        if (startDate > endDate) {
          setTransactions([]);
          setTotal(0);
          setTotalPages(0);
          setSummary({ income: 0, expense: 0, balance: 0 });
          return;
        }

        const query = new URLSearchParams({
          startDate,
          endDate,
          limit: String(limit),
          offset: String(offset),
        });

        if (account !== "all") {
          query.set("account", account);
        }

        if (category !== "all") {
          query.set("category", category);
        }

        if (natureza !== "all") {
          query.set("natureza", natureza);
        }

        const data = await fetchFromAPI<TransactionListResponse>(`/transactions?${query.toString()}`);

        if (!active) {
          return;
        }

        setTransactions(data?.items || []);
        setTotal(data?.total ?? 0);
        setTotalPages(data?.totalPages ?? 0);
        setSummary({
          income: data?.summary?.income ?? 0,
          expense: data?.summary?.expense ?? 0,
          balance: data?.summary?.balance ?? 0,
        });

        if (!data) {
          setError("Não foi possível carregar o extrato.");
        }
      } catch {
        if (!active) {
          return;
        }

        setTransactions([]);
        setTotal(0);
        setTotalPages(0);
        setSummary({ income: 0, expense: 0, balance: 0 });
        setError("Não foi possível carregar o extrato.");
      } finally {
        if (active) {
          setLoading(false);
        }
      }
    }

    loadData();

    return () => {
      active = false;
    };
  }, [account, category, endDate, limit, natureza, offset, startDate]);

  const accountOptions = catalog.accounts;
  const categoryOptions = catalog.categories;
  const natureOptions = catalog.natures.map((nature) => ({
    value: nature,
    label: NATURE_LABELS[nature] ?? nature,
  }));

  const filteredTransactions = useMemo(() => {
    const term = normalizeText(searchTerm);

    return transactions.filter((tx) => {
      const matchesSearch =
        !term ||
        [tx.description, tx.category, tx.natureza, tx.subnatureza, tx.account, tx.date]
          .join(" ")
          .normalize("NFD")
          .replace(/[\u0300-\u036f]/g, "")
          .toUpperCase()
          .includes(term);

      return matchesSearch;
    });
  }, [searchTerm, transactions]);

  const goToPreviousPage = () => {
    if (!hasPreviousPage) {
      return;
    }

    setOffset((current) => Math.max(current - limit, 0));
  };

  const goToNextPage = () => {
    if (!hasNextPage) {
      return;
    }

    setOffset((current) => current + limit);
  };

  const handleStartDateChange = (value: string) => {
    setStartDate(value);
    setOffset(0);
  };

  const handleEndDateChange = (value: string) => {
    setEndDate(value);
    setOffset(0);
  };

  const handleAccountChange = (value: string) => {
    setAccount(value);
    setOffset(0);
  };

  const handleCategoryChange = (value: string) => {
    setCategory(value);
    setOffset(0);
  };

  const handleNaturezaChange = (value: string) => {
    setNatureza(value);
    setOffset(0);
  };

  const openReclassification = (tx: Transaction) => {
    setSelectedTransaction(tx);
    setForm({
      macro_categoria: tx.category,
      natureza: tx.natureza,
      subnatureza: tx.subnatureza,
    });
    setMessage(null);
  };

  void openReclassification;

  const cancelReclassification = () => {
    setSelectedTransaction(null);
    setForm(emptyPayload);
  };

  const handleReclassify = async () => {
    if (!selectedTransaction) {
      return;
    }

    if (!form.macro_categoria.trim() || !form.natureza.trim() || !form.subnatureza.trim()) {
      setMessage("Preencha macro_categoria, natureza e subnatureza.");
      return;
    }

    setSaving(true);
    const result = await fetchFromAPI<{ status: string; message: string }>(
      `/transactions/${selectedTransaction.id}/reclassify`,
      {
        method: "PATCH",
        body: JSON.stringify(form),
      }
    );
    setSaving(false);

    if (!result) {
      setMessage("Não foi possível reclassificar a transação.");
      return;
    }

    setTransactions((current) =>
      current.map((tx) =>
        tx.id === selectedTransaction.id
          ? {
              ...tx,
              category: form.macro_categoria,
              natureza: form.natureza,
              subnatureza: form.subnatureza,
            }
          : tx
      )
    );
    setMessage(result.message);
    cancelReclassification();
  };

  return (
    <PageLayout title="Transações" subtitle="Entrada e saída de capital com paginação real e filtros do backend." hideTitle={true}>
      <section className="rounded-3xl border border-white/10 bg-white/5 p-5 shadow-2xl shadow-black/20 backdrop-blur">
        <div className="flex flex-col gap-4 lg:flex-row lg:items-center lg:justify-between">
          <div>
            <h2 className="mt-2 text-2xl font-semibold tracking-tight text-zinc-50">Base de dados e auditoria</h2>
            <p className="mt-2 max-w-2xl text-sm text-zinc-400">
              Revise e filtre o extrato com paginação real no servidor.
            </p>
          </div>

          <div className="flex items-center gap-3 rounded-2xl border border-white/10 bg-black/20 px-4 py-3 text-sm text-zinc-300">
            <ArrowRightLeft className="size-4 text-cyan-300" />
            <span>{filteredTransactions.length} transações visíveis</span>
          </div>
        </div>

        <div className="mt-6 grid gap-3 md:grid-cols-2 xl:grid-cols-4">
          <article className="rounded-3xl border border-white/10 bg-black/20 p-4">
            <p className="text-xs uppercase tracking-[0.2em] text-zinc-500">Total de Receitas</p>
            <div className="mt-3 flex items-end justify-between gap-3">
              <p className="text-2xl font-semibold tracking-tight text-emerald-300">{formatCurrency(summary.income)}</p>
              <ArrowUpRight className="size-5 text-emerald-300" />
            </div>
          </article>

          <article className="rounded-3xl border border-white/10 bg-black/20 p-4">
            <p className="text-xs uppercase tracking-[0.2em] text-zinc-500">Total de Despesas</p>
            <div className="mt-3 flex items-end justify-between gap-3">
              <p className="text-2xl font-semibold tracking-tight text-rose-300">{formatCurrency(summary.expense)}</p>
              <ArrowDownRight className="size-5 text-rose-300" />
            </div>
          </article>

          <article className="rounded-3xl border border-white/10 bg-black/20 p-4">
            <p className="text-xs uppercase tracking-[0.2em] text-zinc-500">Saldo Líquido</p>
            <div className="mt-3 flex items-end justify-between gap-3">
              <p className={`text-2xl font-semibold tracking-tight ${summary.balance >= 0 ? "text-cyan-300" : "text-rose-300"}`}>
                {formatCurrency(summary.balance)}
              </p>
              <SlidersHorizontal className="size-5 text-cyan-300" />
            </div>
          </article>

          <article className="rounded-3xl border border-white/10 bg-black/20 p-4">
            <p className="text-xs uppercase tracking-[0.2em] text-zinc-500">Movimentações</p>
            <div className="mt-3 flex items-end justify-between gap-3">
              <p className="text-2xl font-semibold tracking-tight text-zinc-50">{filteredTransactions.length}</p>
              <Building2 className="size-5 text-zinc-400" />
            </div>
          </article>
        </div>

        <div className="mt-6 grid gap-3 xl:grid-cols-2">
          <label className="flex items-center gap-3 rounded-2xl border border-white/10 bg-black/20 px-4 py-3 text-zinc-400">
            <Search className="size-4" />
            <input
              value={searchTerm}
              onChange={(event) => setSearchTerm(event.target.value)}
              placeholder="Pesquisar por descrição, categoria, natureza ou conta"
              className="w-full bg-transparent text-sm text-zinc-100 outline-none placeholder:text-zinc-500"
            />
          </label>

          <div className="flex items-center gap-3 rounded-2xl border border-dashed border-white/10 bg-black/10 px-4 py-3 text-sm text-zinc-400">
            <Filter className="size-4" />
            Filtros sincronizados com o backend
          </div>
        </div>

        <div className="mt-3 grid gap-3 lg:grid-cols-2 xl:grid-cols-4">
          <label className="space-y-2 rounded-2xl border border-white/10 bg-black/20 px-4 py-3 text-sm text-zinc-400">
            <span className="flex items-center gap-2 text-xs uppercase tracking-[0.2em] text-zinc-500">

              Período inicial

              value={endDate}

          <div className="mt-4 rounded-2xl border border-cyan-500/20 bg-cyan-500/10 px-4 py-3 text-sm text-cyan-100">

                <X className="mr-2 size-4" />
                Fechar

            </div>

            <div className="mt-5 grid gap-3 md:grid-cols-3">
              <label className="space-y-2 text-sm text-zinc-300">
                <span className="block text-xs uppercase tracking-[0.2em] text-zinc-500">Macro categoria</span>

                  onChange={(event) => setForm((current) => ({ ...current, macro_categoria: event.target.value }))}
                  className="w-full rounded-2xl border border-white/10 bg-black/30 px-4 py-3 text-sm text-zinc-100 outline-none focus:border-emerald-400/50"

              <label className="space-y-2 text-sm text-zinc-300">
                <span className="block text-xs uppercase tracking-[0.2em] text-zinc-500">Natureza</span>
                <input
                  value={form.natureza}
                  onChange={(event) => setForm((current) => ({ ...current, natureza: event.target.value }))}
                  className="w-full rounded-2xl border border-white/10 bg-black/30 px-4 py-3 text-sm text-zinc-100 outline-none focus:border-emerald-400/50"
                />

              <label className="space-y-2 text-sm text-zinc-300">
                <span className="block text-xs uppercase tracking-[0.2em] text-zinc-500">Subnatureza</span>

                ) : (

                    const amountValue = Number(tx.amount) || 0;

                    const kind = classifyTransaction(tx.natureza);

                    // Badge de tipo (linha superior)
                    let typeBadgeClasses: string;
                    let typeBadgeLabel: string;
                    if (kind === "income") {
                      typeBadgeClasses = "bg-green-500/10 text-green-400";

                      .replace(/\b\w/g, (c) => c.toUpperCase());

                    // Formata data DD/MM/YY
                    const dateParts = tx.date.split("-");
                    const formattedDate =
                      dateParts.length === 3
                        ? `${dateParts[2]}/${dateParts[1]}/${dateParts[0].slice(2)}`
                        : tx.date;

                    // Cor do valor
                    let valueColorClass = "text-[#e97070]"; // default expense
                    let valuePrefix = "";
                    if (kind === "income") {
                      valueColorClass = "text-[#4ecf98]";
                      valuePrefix = "+";

              Próxima
              <ChevronRight className="ml-2 size-4" />

    </PageLayout>
  );
}

            { header: "Mês", accessorKey: "month", className: "text-white font-medium" },

function toNumber(value: number | string | null | undefined): number {
  if (typeof value === "number") return value;
  if (typeof value === "string") {
    const parsed = Number.parseFloat(value);
    return Number.isFinite(parsed) ? parsed : 0;
  }
  return 0;
}

function formatDelta(delta: number) {
  const prefix = delta > 0 ? "+" : "";
  return `${prefix}${delta.toFixed(1)}%`;
}


function categoryLabel(category: string) {
  return category.toLowerCase().replace(/_/g, " ");
}

function formatTxDate(value: string) {
  const dt = new Date(`${value}T00:00:00`);
  if (Number.isNaN(dt.getTime())) {
    return value;
  }

  return dt.toLocaleDateString("pt-BR", { day: "2-digit", month: "short" });
}

function getTxVisual(tx: TransactionItem) {
  const natureza = tx.natureza?.toUpperCase();
  if (natureza === "INCOME") {
    return { icon: "↗", box: "bg-green-500/15 text-green-300", value: "text-green-400" };
  }

  if (natureza === "EXPENSE") {
    return { icon: "↘", box: "bg-red-500/15 text-red-300", value: "text-red-400" };
  }

  return { icon: "↔", box: "bg-white/[0.08] text-neutral-300", value: "text-neutral-300" };
}

function LoadingBar({ width = "w-32" }: { width?: string }) {

import { useState, useEffect } from "react";
import { fetchFromAPI } from "@/lib/api";

export interface AssetAllocation {
  asset_class: string;
  current_total: string;
  current_total_value: number;
}

export interface PositionDetail {
  ticker: string;
  asset_class: string;
  asset_subclass: string;
  quantidade: string;
  preco_medio: string;
  valor_investido: string;
  valor_atual: string;
  ganho: string;
  peso_percentual: string;
  quantidade_value?: number;
  preco_medio_value?: number;
  valor_investido_value?: number;
  valor_atual_value: number;
  ganho_value?: number;
  peso_percentual_value: number;
}

export function useAllocationData() {
  const [allocation, setAllocation] = useState<AssetAllocation[]>([]);
  const [positions, setPositions] = useState<PositionDetail[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [isError, setIsError] = useState(false);

  useEffect(() => {
    let mounted = true;
    async function loadData() {
      setIsLoading(true);
      setIsError(false);
      try {
        const [allocRes, posRes] = await Promise.all([
          fetchFromAPI<AssetAllocation[]>("/investments/allocation"),
          fetchFromAPI<PositionDetail[]>("/investments/positions")
        ]);

        if (!mounted) return;

        if (allocRes && posRes) {
          setAllocation(allocRes);
          setPositions(posRes);
        } else {
          setIsError(true);
        }
      } catch {
        if (mounted) setIsError(true);
      } finally {
        if (mounted) setIsLoading(false);
      }
    }
    loadData();
    return () => { mounted = false; };
  }, []);

  const totalAssets = positions.length;
  
  // Sort positions by value desc
  const sortedPositions = [...positions].sort((a, b) => b.valor_atual_value - a.valor_atual_value);
  const top5 = sortedPositions.slice(0, 5);
  
  const totalValue = positions.reduce((acc, p) => acc + p.valor_atual_value, 0);
  const top5Value = top5.reduce((acc, p) => acc + p.valor_atual_value, 0);
  const top5Concentration = totalValue > 0 ? (top5Value / totalValue) * 100 : 0;

  let dominantClass = { name: "N/A", percentage: 0 };
  if (allocation.length > 0 && totalValue > 0) {
    const sortedAlloc = [...allocation].sort((a, b) => b.current_total_value - a.current_total_value);
    dominantClass = {
      name: sortedAlloc[0].asset_class,
      percentage: (sortedAlloc[0].current_total_value / totalValue) * 100
    };
  }

  // Very simple diversification score: penalizes high top-5 concentration and rewards number of assets
  const calcScore = () => {
    if (totalAssets === 0) return 0;
    let score = 100;
    // penalty for top 5 concentration > 30%
    if (top5Concentration > 30) {
      score -= (top5Concentration - 30) * 1.5; 
    }
    // penalty for few assets
    if (totalAssets < 10) score -= (10 - totalAssets) * 2;
    return Math.max(0, Math.min(100, Math.round(score)));
  };

  const diversificationScore = calcScore();

  return {
    allocation,
    positions: sortedPositions,
    top5,
    top5Concentration,
    totalAssets,
    dominantClass,
    diversificationScore,
    isLoading,
    isError
  };
}

import { useState, useEffect } from "react";
import { fetchFromAPI } from "@/lib/api";
import { PortfolioEvolutionPoint, PortfolioEvolutionSummary } from "./useGrowthData";

export function useAlphaData(period: string) {
  const [history, setHistory] = useState<PortfolioEvolutionPoint[]>([]);
  const [summary, setSummary] = useState<PortfolioEvolutionSummary | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [isError, setIsError] = useState(false);

  useEffect(() => {
    let mounted = true;
    async function loadData() {
      setIsLoading(true);
      setIsError(false);
      try {
        const [histRes, sumRes] = await Promise.all([
          fetchFromAPI<PortfolioEvolutionPoint[]>("/investments/evolution/history"),
          fetchFromAPI<PortfolioEvolutionSummary>("/investments/evolution/summary")
        ]);

        if (!mounted) return;
        if (histRes) setHistory(histRes);
        if (sumRes) setSummary(sumRes);
        if (!histRes || !sumRes) setIsError(true);
      } catch {
        if (mounted) setIsError(true);
      } finally {
        if (mounted) setIsLoading(false);
      }
    }
    loadData();
    return () => { mounted = false; };
  }, []);

  let filteredHistory = history;
  if (period !== "ALL" && history.length > 0) {
    let monthsToKeep = history.length;
    switch (period) {
      case "3M": monthsToKeep = 3; break;
      case "6M": monthsToKeep = 6; break;
      case "1A": monthsToKeep = 12; break;
      case "2A": monthsToKeep = 24; break;
    }
    filteredHistory = history.slice(Math.max(history.length - monthsToKeep, 0));
  }

  // Lógica de Rebase: Subtrair o acumulado do primeiro mês do período
  // para que todas as linhas comecem do zero relativo ao período filtrado.
  const rebasedHistory = filteredHistory.map((point, index) => {
    if (index === 0 || !filteredHistory[0]) return { ...point, rebased_cart: 0, rebased_ibov: 0, rebased_cdi: 0, rebased_ipca: 0, rebased_ifix: 0, rebased_sp500: 0, rebased_idiv: 0 };
    
    const base = filteredHistory[0];
    return {
      ...point,
      rebased_cart: point.rentabilidade_acumulada_pct - base.rentabilidade_acumulada_pct,
      rebased_ibov: point.ibov_acumulado_pct - base.ibov_acumulado_pct,
      rebased_cdi: point.cdi_acumulado_pct - base.cdi_acumulado_pct,
      rebased_ipca: point.ipca_acumulado_pct - base.ipca_acumulado_pct,
      rebased_ifix: point.ifix_acumulado_pct - base.ifix_acumulado_pct,
      rebased_sp500: point.sp500_acumulado_pct - base.sp500_acumulado_pct,
      rebased_idiv: point.idiv_acumulado_pct - base.idiv_acumulado_pct
    };
  });

  return {
    history: rebasedHistory,
    rawHistory: filteredHistory,
    summary,
    isLoading,
    isError
  };
}

==== src/hooks/useDividendsData.ts ====
import { useState, useEffect } from "react";
import { fetchFromAPI } from "@/lib/api";
import { PortfolioEvolutionSummary } from "./useGrowthData";

export interface DividendHistory {
  month: string; // e.g. "2023-01"
  amount_received: string;
  amount_received_value: number;
  amount_dividend_value?: number;
  amount_jcp_value?: number;
  amount_rendimento_value?: number;
}

export interface DividendBreakdown {
  ticker: string;
  total_recebido: string;
  total_recebido_value: number;
  quantidade_eventos: number;
  primeiro_recebimento: string;
  ultimo_recebimento: string;
}

export function useDividendsData() {
  const [history, setHistory] = useState<DividendHistory[]>([]);
  const [breakdown, setBreakdown] = useState<DividendBreakdown[]>([]);
  const [summary, setSummary] = useState<PortfolioEvolutionSummary | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [isError, setIsError] = useState(false);

  useEffect(() => {
    let mounted = true;
    async function loadData() {
      setIsLoading(true);
      setIsError(false);
      try {
        const [histRes, sumRes, breakRes] = await Promise.all([
          fetchFromAPI<DividendHistory[]>("/investments/dividends/history"),
          fetchFromAPI<PortfolioEvolutionSummary>("/investments/evolution/summary"),
          fetchFromAPI<DividendBreakdown[]>("/investments/dividends/breakdown")
        ]);

        if (!mounted) return;
        if (histRes) {
          setHistory(histRes);
          setSummary(sumRes || null);
          setBreakdown(breakRes || []);
        } else {
          setIsError(true);
        }
      } catch {
        if (mounted) setIsError(true);
      } finally {
        if (mounted) setIsLoading(false);
      }
    }
    loadData();
    return () => { mounted = false; };
  }, []);

  const totalDividends = summary?.proventos_totais || 0;
  
  // Calculate 12-month average
  let average12m = 0;
  if (history.length > 0) {
    const last12 = history.slice(Math.max(history.length - 12, 0));
    const sum12m = last12.reduce((acc, curr) => acc + (curr.amount_received_value || 0), 0);
    average12m = sum12m / (last12.length || 1);
  }

  // Calculate yield on cost
  const yieldOnCost = summary?.compras_totais && summary.compras_totais > 0 
    ? (totalDividends / summary.compras_totais) * 100 
    : 0;

  // Calculate YTD received (current calendar year)
  const currentYear = new Date().getFullYear().toString();
  const ytdReceived = history
    .filter((h) => h.month?.startsWith(currentYear))
    .reduce((acc, h) => acc + (h.amount_received_value || 0), 0);

  // Transform for YoY Chart: group by month (Jan-Dec) and year
  // Result format: { name: 'Jan', '2023': 10, '2024': 15 }
  const yoyData: Record<string, Record<string, number | string>> = {};
  const months = ['Jan', 'Fev', 'Mar', 'Abr', 'Mai', 'Jun', 'Jul', 'Ago', 'Set', 'Out', 'Nov', 'Dez'];
  
  months.forEach(m => {
    yoyData[m] = { name: m };
  });

  const availableYears = new Set<string>();

  history.forEach(item => {
    if (!item.month) return;
    const [year, monthStr] = item.month.split('-');
    if (year && monthStr) {
      const monthIdx = parseInt(monthStr, 10) - 1;
      const monthName = months[monthIdx];
      availableYears.add(year);
      yoyData[monthName][year] = item.amount_received_value;
    }
  });

  const yoyArray = months.map(m => yoyData[m]);

  return {
    history,
    breakdown,
    yoyArray,
    availableYears: Array.from(availableYears).sort(),
    totalDividends,
    average12m,
    yieldOnCost,
    ytdReceived,
    isLoading,
    isError
  };
}

import { useState, useEffect } from "react";
import { fetchFromAPI } from "@/lib/api";
// Let's define local types based on the backend schema.

export interface PortfolioEvolutionSummary {
  patrimonio_atual: string;
  patrimonio_atual_value: number;
  rentabilidade_acumulada: number;
  proventos_totais: number;

  cdi_acumulado_pct: number;
  alpha_vs_ibov: number;
  alpha_vs_cdi: number;

export interface PortfolioEvolutionPoint {
  month: string;
  patrimonio: string;
  patrimonio_value: number;
  compras: string;
  compras_value: number;
  vendas: string;
  vendas_value: number;
  proventos: string;
  proventos_value: number;
  rentabilidade_mensal_pct: number;
  rentabilidade_acumulada_pct: number;
  ibov_mensal_pct: number;
  ibov_acumulado_pct: number;
  cdi_mensal_pct: number;
  cdi_acumulado_pct: number;
  ipca_mensal_pct: number;
  ipca_acumulado_pct: number;
  ifix_mensal_pct: number;
  ifix_acumulado_pct: number;
  sp500_mensal_pct: number;
  sp500_acumulado_pct: number;
  idiv_mensal_pct: number;
  idiv_acumulado_pct: number;
}

export function useGrowthData(period: string) {
  const [summary, setSummary] = useState<PortfolioEvolutionSummary | null>(null);
  const [history, setHistory] = useState<PortfolioEvolutionPoint[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [isError, setIsError] = useState(false);

  useEffect(() => {
    let mounted = true;

    async function loadData() {
      setIsLoading(true);
      setIsError(false);
      try {
        const [sumRes, histRes] = await Promise.all([
          fetchFromAPI<PortfolioEvolutionSummary>("/investments/evolution/summary"),
          fetchFromAPI<PortfolioEvolutionPoint[]>("/investments/evolution/history")
        ]);

        if (!mounted) return;

        if (sumRes && histRes) {
          setSummary(sumRes);
          setHistory(histRes);
        } else {
          setIsError(true);
        }
      } catch {
        if (mounted) setIsError(true);
      } finally {
        if (mounted) setIsLoading(false);
      }
    }

    loadData();

    return () => {
      mounted = false;
    };
  }, []);

  // Filter history based on period
  let filteredHistory = history;
  if (period !== "ALL" && history.length > 0) {
    let monthsToKeep = history.length;
    switch (period) {
      case "3M": monthsToKeep = 3; break;
      case "6M": monthsToKeep = 6; break;
      case "1A": monthsToKeep = 12; break;
      case "2A": monthsToKeep = 24; break;
    }
    filteredHistory = history.slice(Math.max(history.length - monthsToKeep, 0));
  }

  // Calculate milestones
  const milestones: { value: number; label: string; date: string }[] = [];
  if (history.length > 0) {
    const checkMilestone = (value: number, label: string) => {
      const point = history.find(p => p.patrimonio_value >= value);
      if (point) {
        milestones.push({ value, label, date: point.month });
        return true;
      }
      return false;
    };
    checkMilestone(10000, "Primeiros R$ 10k");
    checkMilestone(50000, "R$ 50k Acumulados");
    checkMilestone(100000, "R$ 100k Alcançados");
    
    let nextCheck = 150000;
    while (checkMilestone(nextCheck, `R$ ${nextCheck / 1000}k Alcançados`)) {
      nextCheck += 50000;
    }
  }

  let nextTarget = 10000;
  const pat = summary?.patrimonio_atual_value || 0;
  if (pat < 10000) nextTarget = 10000;
  else if (pat < 50000) nextTarget = 50000;
  else if (pat < 100000) nextTarget = 100000;
  else {
    nextTarget = Math.floor(pat / 50000) * 50000 + 50000;
  }

  return {
    summary,
    history: filteredHistory,
    milestones,
    nextTarget,
    isLoading,
    isError
  };
}
