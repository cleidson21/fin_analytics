VIEWS
============================================================
  vw_rules_training_dataset
    CREATE VIEW vw_rules_training_dataset AS
    SELECT 
        t.id_economico,
        t.data,
        t.descricao_original,
        t.descricao_normalizada,
        t.valor_cents,
        t.macro_categoria,
        t.sub_categoria,
        t.natureza,
        t.classificacao_fonte,
        t.applied_rule_id,
        r.rule_type,
        r.confidence_score,
        r.evidence_count,
        r.reason
    FROM transactions t
    LEFT JOIN rules r ON t.applied_rule_id = r.id
